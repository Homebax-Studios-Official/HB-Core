# Test backend: system/downloaded GoogleTest if available, otherwise a tiny built-in
# GoogleTest-compatible harness (tests/mini_gtest.h) so tests still run offline.
set(HB_TEST_BACKEND "mini")
find_package(GTest QUIET)
if(GTest_FOUND)
    set(HB_TEST_BACKEND "gtest")
    set(HB_GTEST_LIBS GTest::gtest GTest::gtest_main)
elseif(HBCORE_FETCH_GTEST)
    set(_gt_zip "${CMAKE_BINARY_DIR}/googletest-1.15.2.zip")
    if(NOT EXISTS "${_gt_zip}")
        file(DOWNLOAD "https://github.com/google/googletest/archive/refs/tags/v1.15.2.zip" "${_gt_zip}"
            STATUS _gt_status TIMEOUT 60 TLS_VERIFY ON)
        list(GET _gt_status 0 _gt_code)
        if(NOT _gt_code EQUAL 0)
            file(REMOVE "${_gt_zip}")
        endif()
    endif()
    if(EXISTS "${_gt_zip}")
        include(FetchContent)
        set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
        set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
        FetchContent_Declare(googletest URL "${_gt_zip}" DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
        FetchContent_MakeAvailable(googletest)
        set(HB_TEST_BACKEND "gtest")
        set(HB_GTEST_LIBS gtest gtest_main)
    endif()
endif()
message(STATUS "HB Core test backend: ${HB_TEST_BACKEND}")

function(hbcore_test name)
    cmake_parse_arguments(T "" "" "SOURCES;DEPS" ${ARGN})
    if(HB_TEST_BACKEND STREQUAL "gtest")
        add_executable(${name} ${T_SOURCES})
        target_link_libraries(${name} PRIVATE ${T_DEPS} ${HB_GTEST_LIBS})
        target_compile_definitions(${name} PRIVATE HB_TEST_USE_GTEST=1)
    else()
        add_executable(${name} ${T_SOURCES} mini_main.cpp)
        target_link_libraries(${name} PRIVATE ${T_DEPS})
    endif()
    target_include_directories(${name} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    add_test(NAME ${name} COMMAND ${name} WORKING_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR})
endfunction()
